---
title: SQLite bindings
description: Node, Bun, Ruby, Go, and Elixir bindings over Knocker's shared SQLite contract.
---

Knocker is a loadable SQLite extension plus language bindings. The bindings
on this page talk to the same shared SQLite contract as the Python package.

Every binding in this page proves the same baseline:

- open/bootstrap
- endpoint registration
- ingest
- endpoint handler registration plus one worker-processing path
- `get_event`
- `list_events`
- `replay`

These bindings load the Knocker SQLite extension directly. They do not wrap
separate Honker language packages; queue operations go through the shared
SQLite functions exposed by the extension. They are not framework adapters.

## Node

Source: `packages/knocker-node`

```js
import { KnockerSqlite } from "./packages/knocker-node/index.mjs";

const db = new KnockerSqlite("app.db");
db.addEndpoint({
  name: "github",
  path: "/webhooks/github",
  provider: "github",
});

db.ingest({
  endpoint: "github",
  body: Buffer.from('{"hello":"world"}'),
  providerDeliveryId: "bun-delivery-1",
  signatureValid: true,
});

const result = db.ingest({
  endpoint: "github",
  body: Buffer.from('{"hello":"world"}'),
  providerDeliveryId: "node-delivery-1",
  signatureValid: true,
});

db.registerHandler("github", (event) => {
  console.log("handled", event.id);
});

db.runWorkerOnce("node-worker");
```

## Ruby

Source: `packages/knocker-ruby`

```ruby
require_relative "../packages/knocker-ruby/lib/knocker_sqlite"

db = KnockerSqlite::Database.open("app.db")
db.add_endpoint(
  name: "github",
  path: "/webhooks/github",
  provider: "github",
  enabled: true,
)

db.ingest(
  endpoint: "github",
  body: "{\"hello\":\"world\"}",
  provider_delivery_id: "ruby-delivery-1",
  signature_valid: true,
)

db.register_handler("github") do |event|
  puts "handled #{event['id']}"
end

db.run_worker_once(worker_id: "ruby-worker")
```

## Go

Source: `packages/knocker-go`

```go
webhooks, err := knockersqlite.Open("app.db")
if err != nil {
    panic(err)
}
defer webhooks.Close()

provider := "github"
_, err = webhooks.AddEndpoint(knockersqlite.AddEndpointParams{
    Name:     "github",
    Path:     "/webhooks/github",
    Provider: &provider,
    Enabled:  true,
})

webhooks.RegisterHandler("github", func(event knockersqlite.Event) error {
    return nil
})

deliveryID := "go-delivery-1"
signatureValid := true
_, err = webhooks.Ingest(knockersqlite.IngestParams{
    Endpoint:           "github",
    Body:               []byte(`{"hello":"world"}`),
    ProviderDeliveryID: &deliveryID,
    SignatureValid:     &signatureValid,
})

_, err = webhooks.RunWorkerOnce("go-worker")
```

## Elixir

Source: `packages/knocker-elixir`

```elixir
{:ok, opened} = KnockerSqlite.open("app.db")

webhooks =
  KnockerSqlite.register_handler(opened, "github", fn event ->
    IO.inspect(event["id"], label: "handled")
  end)

{:ok, _endpoint_id} =
  KnockerSqlite.add_endpoint(webhooks, %{
    name: "github",
    path: "/webhooks/github",
    provider: "github",
    enabled: true
  })

{:ok, _result} =
  KnockerSqlite.ingest(webhooks, %{
    endpoint: "github",
    body: ~s({"hello":"world"}),
    provider_delivery_id: "elixir-delivery-1",
    signature_valid: true
  })

{:ok, _event_id} = KnockerSqlite.run_worker_once(webhooks, "elixir-worker")
```
